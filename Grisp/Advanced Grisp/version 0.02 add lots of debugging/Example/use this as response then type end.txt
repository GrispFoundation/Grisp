⟦BEGIN PROGRAM⟧
  ⟦BEGIN METADATA⟧plan_id = "go2delphi-manual"⟦END METADATA⟧

  ⟦BEGIN VARIABLE⟧go_source⟦END VARIABLE⟧
  ⟦BEGIN VARIABLE⟧ast⟦END VARIABLE⟧
  ⟦BEGIN VARIABLE⟧delphi_code⟦END VARIABLE⟧

  ⟦CALL FileSystem.read_file(path = WorldState.go_file) INTO go_source⟧

  ⟦IF go_source != empty THEN⟧
    ⟦CALL GoParser.parse(source_code = go_source) INTO ast⟧
    ⟦CALL DelphiGenerator.generate(ast = ast) INTO delphi_code⟧
    ⟦CALL FileSystem.write_file(path = WorldState.pas_file, content = delphi_code)⟧
  ⟦ELSE⟧
    ⟦BEGIN ON FAILURE⟧
      ⟦BEGIN FAILURE CONDITION⟧precondition_failed⟧END FAILURE CONDITION⟧
      ⟦BEGIN FALLBACK⟧
        ⟦CALL Logger.write(message = "Source file not found")⟧
      ⟦END FALLBACK⟧
    ⟦END ON FAILURE⟧
  ⟦END IF⟧
⟦END PROGRAM⟧
