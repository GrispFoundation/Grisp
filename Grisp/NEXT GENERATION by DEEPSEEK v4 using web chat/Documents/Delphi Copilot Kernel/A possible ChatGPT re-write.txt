files["GSDA31.Canonical.pas"] = r'''unit GSDA31.Canonical;

interface

uses
    System.SysUtils,
    System.JSON,
    System.Classes,
    System.Generics.Collections;

type
    TGSDAJsonCanonical = class
    private
        class function CompareUtf8(const ALeft, ARight: string): Integer; static;
        class procedure AppendEscapedString(const AValue: string; const ABuilder: TStringBuilder); static;
        class procedure AppendObject(const AObject: TJSONObject; const ABuilder: TStringBuilder); static;
        class procedure AppendArray(const AArray: TJSONArray; const ABuilder: TStringBuilder); static;
        class procedure AppendNumber(const ANumber: TJSONNumber; const ABuilder: TStringBuilder); static;
        class procedure AppendValue(const AValue: TJSONValue; const ABuilder: TStringBuilder); static;
    public
        class function Canonicalize(const AValue: TJSONValue): string; static;
    end;

implementation

class function TGSDAJsonCanonical.CompareUtf8(const ALeft, ARight: string): Integer;
var
    LBytes: TBytes;
    RBytes: TBytes;
    LIndex: Integer;
    LLength: Integer;
begin
    LBytes := TEncoding.UTF8.GetBytes(ALeft);
    RBytes := TEncoding.UTF8.GetBytes(ARight);

    LLength := Length(LBytes);
    if Length(RBytes) < LLength then
        LLength := Length(RBytes);

    for LIndex := 0 to LLength - 1 do
    begin
        if LBytes[LIndex] < RBytes[LIndex] then
            Exit(-1);
        if LBytes[LIndex] > RBytes[LIndex] then
            Exit(1);
    end;

    if Length(LBytes) < Length(RBytes) then
        Result := -1
    else if Length(LBytes) > Length(RBytes) then
        Result := 1
    else
        Result := 0;
end;

class procedure TGSDAJsonCanonical.AppendEscapedString(
    const AValue: string;
    const ABuilder: TStringBuilder);
var
    LIndex: Integer;
    LCode: Integer;
    LChar: Char;
begin
    ABuilder.Append('"');

    for LIndex := 1 to Length(AValue) do
    begin
        LChar := AValue[LIndex];
        LCode := Ord(LChar);

        case LChar of
            '"': ABuilder.Append('\"');
            '\': ABuilder.Append('\\');
            #8: ABuilder.Append('\b');
            #9: ABuilder.Append('\t');
            #10: ABuilder.Append('\n');
            #12: ABuilder.Append('\f');
            #13: ABuilder.Append('\r');
        else
            if LCode < 32 then
                ABuilder.Append('\u').Append(IntToHex(LCode, 4))
            else
                ABuilder.Append(LChar);
        end;
    end;

    ABuilder.Append('"');
end;

class procedure TGSDAJsonCanonical.AppendNumber(
    const ANumber: TJSONNumber;
    const ABuilder: TStringBuilder);
var
    LText: string;
    LIndex: Integer;
    LHasExponent: Boolean;
begin
    LText := ANumber.ToString;

    if LText = '' then
        raise EGSDAFailure.Create(fcSchemaFailure, 'Empty JSON number');

    for LIndex := 1 to Length(LText) do
    begin
        if not CharInSet(LText[LIndex], ['0'..'9', '-', '+', '.', 'e', 'E']) then
            raise EGSDAFailure.Create(fcSchemaFailure, 'Invalid JSON number');
    end;

    if Pos('NaN', LText) > 0 then
        raise EGSDAFailure.Create(fcSchemaFailure, 'NaN not permitted');

    if Pos('Infinity', LText) > 0 then
        raise EGSDAFailure.Create(fcSchemaFailure, 'Infinity not permitted');

    if SameText(LText, '-0') or SameText(LText, '-0.0') then
    begin
        ABuilder.Append('0');
        Exit;
    end;

    LHasExponent := (Pos('e', LowerCase(LText)) > 0);

    if not LHasExponent then
        ABuilder.Append(LText)
    else
        ABuilder.Append(LText);
end;

class procedure TGSDAJsonCanonical.AppendObject(
    const AObject: TJSONObject;
    const ABuilder: TStringBuilder);
var
    LPairs: TList<TJSONPair>;
    LSeen: TDictionary<string, Boolean>;
    LPair: TJSONPair;
    LPairIndex: Integer;
begin
    LPairs := TList<TJSONPair>.Create;
    LSeen := TDictionary<string, Boolean>.Create;
    try
        for LPair in AObject do
        begin
            if LSeen.ContainsKey(LPair.JsonString.Value) then
                raise EGSDAFailure.Create(
                    fcSchemaFailure,
                    'Duplicate JSON object key: ' + LPair.JsonString.Value
                );

            LSeen.Add(LPair.JsonString.Value, True);
            LPairs.Add(LPair);
        end;

        LPairs.Sort(
            TComparer<TJSONPair>.Construct(
                function(const ALeft, ARight: TJSONPair): Integer
                begin
                    Result := CompareUtf8(
                        ALeft.JsonString.Value,
                        ARight.JsonString.Value
                    );
                end
            )
        );

        ABuilder.Append('{');

        for LPairIndex := 0 to LPairs.Count - 1 do
        begin
            AppendEscapedString(LPairs[LPairIndex].JsonString.Value, ABuilder);
            ABuilder.Append(':');
            AppendValue(LPairs[LPairIndex].JsonValue, ABuilder);

            if LPairIndex < LPairs.Count - 1 then
                ABuilder.Append(',');
        end;

        ABuilder.Append('}');
    finally
        LSeen.Free;
        LPairs.Free;
    end;
end;

class procedure TGSDAJsonCanonical.AppendArray(
    const AArray: TJSONArray;
    const ABuilder: TStringBuilder);
var
    LIndex: Integer;
begin
    ABuilder.Append('[');

    for LIndex := 0 to AArray.Count - 1 do
    begin
        AppendValue(AArray.Items[LIndex], ABuilder);

        if LIndex < AArray.Count - 1 then
            ABuilder.Append(',');
    end;

    ABuilder.Append(']');
end;

class procedure TGSDAJsonCanonical.AppendValue(
    const AValue: TJSONValue;
    const ABuilder: TStringBuilder);
begin
    if AValue = nil then
        raise EGSDAFailure.Create(fcSchemaFailure, 'Missing JSON value');

    if AValue is TJSONObject then
        AppendObject(TJSONObject(AValue), ABuilder)
    else if AValue is TJSONArray then
        AppendArray(TJSONArray(AValue), ABuilder)
    else if AValue is TJSONString then
        AppendEscapedString(TJSONString(AValue).Value, ABuilder)
    else if AValue is TJSONNumber then
        AppendNumber(TJSONNumber(AValue), ABuilder)
    else if AValue is TJSONBool then
        ABuilder.Append(LowerCase(AValue.ToString))
    else if AValue is TJSONNull then
        ABuilder.Append('null')
    else
        raise EGSDAFailure.Create(fcSchemaFailure, 'Unsupported JSON node');
end;

class function TGSDAJsonCanonical.Canonicalize(const AValue: TJSONValue): string;
var
    LBuilder: TStringBuilder;
begin
    if AValue = nil then
        raise EGSDAFailure.Create(fcSchemaFailure, 'Cannot canonicalize nil');

    LBuilder := TStringBuilder.Create;
    try
        AppendValue(AValue, LBuilder);
        Result := LBuilder.ToString;
    finally
        LBuilder.Free;
    end;
end;

end.
'''

files["GSDA31.Crypto.pas"] = r'''unit GSDA31.Crypto;

interface

uses
    System.SysUtils,
    System.Hash,
    System.DateUtils;

function GSDA_SHA256Hex(const AValue: string): string;
function GSDA_HMACSHA256Hex(const ASecret: TBytes; const AValue: string): string;
function GSDA_UtcNowIso: string;
function GSDA_ConstantTimeEquals(const ALeft, ARight: string): Boolean;

implementation

function GSDA_SHA256Hex(const AValue: string): string;
begin
    Result := THashSHA2.GetHashString(TEncoding.UTF8.GetBytes(AValue));
end;

function GSDA_HMACSHA256Hex(const ASecret: TBytes; const AValue: string): string;
var
    LHmac: THashSHA2;
    LMessage: TBytes;
begin
    LMessage := TEncoding.UTF8.GetBytes(AValue);
    LHmac := THashSHA2.Create;
    try
        Result := LHmac.GetHMACAsString(LMessage, ASecret, SHA256);
    finally
        LHmac.Free;
    end;
end;

function GSDA_UtcNowIso: string;
begin
    Result := DateToISO8601(TTimeZone.Local.ToUniversalTime(Now), True);
end;

function GSDA_ConstantTimeEquals(const ALeft, ARight: string): Boolean;
var
    LIndex: Integer;
    LDiff: Integer;
    LBytesLeft: TBytes;
    LBytesRight: TBytes;
begin
    LBytesLeft := TEncoding.UTF8.GetBytes(ALeft);
    LBytesRight := TEncoding.UTF8.GetBytes(ARight);

    LDiff := Length(LBytesLeft) xor Length(LBytesRight);

    for LIndex := 0 to Length(LBytesLeft) - 1 do
        if LIndex < Length(LBytesRight) then
            LDiff := LDiff or (LBytesLeft[LIndex] xor LBytesRight[LIndex])
        else
            LDiff := LDiff or LBytesLeft[LIndex];

    for LIndex := 0 to Length(LBytesRight) - 1 do
        if LIndex >= Length(LBytesLeft) then
            LDiff := LDiff or LBytesRight[LIndex];

    Result := LDiff = 0;
end;

end.
'''

files["GSDA31.BigInt.pas"] = r'''unit GSDA31.BigInt;

interface

uses
    System.SysUtils;

type
    TGSDAUInt128 = record
        High: UInt64;
        Low: UInt64;
        class function Zero: TGSDAUInt128; static;
        class function FromUInt64(AValue: UInt64): TGSDAUInt128; static;
        function IsZero: Boolean;
        class operator Add(const ALeft, ARight: TGSDAUInt128): TGSDAUInt128;
        class operator Subtract(const ALeft, ARight: TGSDAUInt128): TGSDAUInt128;
        class operator Multiply(const ALeft, ARight: TGSDAUInt128): TGSDAUInt128;
        class operator Implicit(AValue: UInt64): TGSDAUInt128;
        function ToDecimalString: string;
    end;

implementation

class function TGSDAUInt128.Zero: TGSDAUInt128;
begin
    Result.High := 0;
    Result.Low := 0;
end;

class function TGSDAUInt128.FromUInt64(AValue: UInt64): TGSDAUInt128;
begin
    Result.High := 0;
    Result.Low := AValue;
end;

function TGSDAUInt128.IsZero: Boolean;
begin
    Result := (High = 0) and (Low = 0);
end;

class operator TGSDAUInt128.Add(
    const ALeft, ARight: TGSDAUInt128): TGSDAUInt128;
var
    LCarry: UInt64;
begin
    Result.Low := ALeft.Low + ARight.Low;
    LCarry := Ord(Result.Low < ALeft.Low);

    if UInt64(ALeft.High) > High(UInt64) - UInt64(ARight.High) - LCarry then
        raise EOverflow.Create('UInt128 addition overflow');

    Result.High := ALeft.High + ARight.High + LCarry;
end;

class operator TGSDAUInt128.Subtract(
    const ALeft, ARight: TGSDAUInt128): TGSDAUInt128;
var
    LBorrow: UInt64;
begin
    if (ALeft.High < ARight.High) or
       ((ALeft.High = ARight.High) and (ALeft.Low < ARight.Low)) then
        raise ERangeError.Create('UInt128 subtraction underflow');

    LBorrow := Ord(ALeft.Low < ARight.Low);
    Result.Low := ALeft.Low - ARight.Low;
    Result.High := ALeft.High - ARight.High - LBorrow;
end;

class operator TGSDAUInt128.Multiply(
    const ALeft, ARight: TGSDAUInt128): TGSDAUInt128;
var
    LOldHigh, LPart1, LPart2, LPart3, LCarry: UInt64;
begin
    { Exact result is only retained when it fits into 128 bits.
      This implementation uses 32-bit limbs to avoid compiler-specific 128-bit types. }
    LPart1 := (ALeft.Low and $FFFFFFFF) * (ARight.Low and $FFFFFFFF);
    LPart2 := (ALeft.Low shr 32) * (ARight.Low and $FFFFFFFF);
    LPart3 := (ALeft.Low and $FFFFFFFF) * (ARight.Low shr 32);

    Result.Low := LPart1 + ((LPart2 + LPart3) shl 32);
    LCarry := Ord(Result.Low < LPart1);

    LOldHigh := ALeft.High * ARight.Low + ALeft.Low * ARight.High;
    if LOldHigh < ALeft.High * ARight.Low then
        raise EOverflow.Create('UInt128 multiplication overflow');

    Result.High := LOldHigh + (LPart2 shr 32) + (LPart3 shr 32) + LCarry;

    if Result.High < LOldHigh then
        raise EOverflow.Create('UInt128 multiplication overflow');
end;

class operator TGSDAUInt128.Implicit(AValue: UInt64): TGSDAUInt128;
begin
    Result := FromUInt64(AValue);
end;

function TGSDAUInt128.ToDecimalString: string;
var
    LHigh: UInt64;
    LLow: UInt64;
    LPart: UInt64;
    LDigits: string;
begin
    { Decimal conversion without relying on compiler-specific UInt128 support. }
    if IsZero then
        Exit('0');

    LHigh := High;
    LLow := Low;
    LDigits := '';

    while (LHigh <> 0) or (LLow <> 0) do
    begin
        LPart := LLow mod 10;
        LLow := LLow div 10;

        if (LHigh <> 0) then
        begin
            LLow := LLow + (LHigh mod 10) shl 63;
            LHigh := LHigh div 10;
        end;

        LDigits := Char(Ord('0') + Integer(LPart mod 10)) + LDigits;
    end;

    Result := LDigits;
end;

end.
'''

    const AValues: TArray<string>): TArray<string>;

    LList: TList<string>;
    LValue: string;

    LList := TList<string>.Create;
    try
        for LValue in AValues do
            LList.Add(LValue);

        LList.Sort(
            TComparer<string>.Construct(
                function(const ALeft, ARight: string): Integer
                begin
                    Result := CompareStr(ALeft, ARight);
                end
            )
        );

        Result := LList.ToArray;
    finally
        LList.Free;
    end;

    const AObject: TJSONObject;

    const AValues: TArray<string>);

    LArray: TJSONArray;
    LValue: string;

    LArray := TJSONArray.Create;
    for LValue in SortedStringArray(AValues) do
        LArray.Add(LValue);

    AObject.AddPair(AName, LArray);

        LWrapper.Free;
    end;

function TGSDAKernel31.ParsePayload(const ARawPayload: string): TJSONObject;
var
    LSanitized: string;
    LValue: TJSONValue;
begin
    LSanitized := Trim(ARawPayload);

    if (Length(LSanitized) >= 3) and
       (LSanitized[1] = #$EF) and
       (LSanitized[2] = #$BB) and
       (LSanitized[3] = #$BF) then
        Delete(LSanitized, 1, 3);

    LValue := TJSONObject.ParseJSONValue(LSanitized);
    if not (LValue is TJSONObject) then
    begin
        LValue.Free;
        raise EGSDAFailure.Create(fcProtocolFailure, 'Payload is not a JSON object');
    end;

    Result := TJSONObject(LValue);
end;

function TGSDAKernel31.ComputeContentID(const APayload: TJSONObject): string;
var
    LContentProjection: TJSONObject;
begin
    LContentProjection := APayload.Clone as TJSONObject;
    try
        LContentProjection.RemovePair('message_id').Free;
        LContentProjection.RemovePair('hmac').Free;
        LContentProjection.RemovePair('peer_id').Free;
        LContentProjection.RemovePair('run_id').Free;
        LContentProjection.RemovePair('round_id').Free;
        LContentProjection.RemovePair('logical_sequence').Free;

        Result := 'content:' +
            GSDA_SHA256Hex(
                TGSDAJsonCanonical.Canonicalize(LContentProjection)
            );
    finally
        LContentProjection.Free;
    end;
end;

    const APeer: TGSDAPeerProfile;
    const AFingerprint, ARoundID, AToken: string): Boolean;

    LFingerprint: string;
    LExpected: string;
begin
    LFingerprint :=
        APeer.TransportName + '|' +
        APeer.AdapterID + '|' +
        APeer.EndpointFingerprint;

    if not GSDA_ConstantTimeEquals(LFingerprint, AFingerprint) then
        Exit(False);

    LExpected := GSDA_HMACSHA256Hex(
        APeer.SharedSecret,
        APeer.PeerID + '|' + AFingerprint + '|' + ARoundID
    );

    Result := GSDA_ConstantTimeEquals(LExpected, AToken);
end;

function TGSDAKernel31.CanonicalArrayOfIds(
    const AIds: TArray<string>): TJSONArray;
var
    LList: TList<string>;
    LValue: string;
begin
    LList := TList<string>.Create;
    try
        for LValue in AIds do
            LList.Add(LValue);

        LList.Sort;

        Result := TJSONArray.Create;
        for LValue in LList do
            Result.Add(LValue);
    finally
        LList.Free;
    end;
end;

function TGSDAKernel31.AcceptPeerMessage(
    const APeerID, ARunID, ATaskID, ARoundID, AToken, AFingerprint: string;
    AKind: TGSDAMessageKind;
    AConfidence: Integer;
    const ARawPayload: string): TGSDAIngressResult;
var
    LPeer: TGSDAPeerProfile;
    LRunTask: TGSDAResearchTask;
    LPayload: TJSONObject;
    LCanonical: string;
    LContentID: string;
    LPayloadSchema: string;
    LMessageID: string;
    LSequence: Int64;
    LMessage: TGSDAReasoningMessage;
    LIdentityPreimage: TJSONObject;
begin
    FillChar(Result, SizeOf(Result), 0);

    ValidatePeer(APeerID);

    if not FPeers.TryGetValue(APeerID, LPeer) then
        raise EGSDAFailure.Create(fcImpersonationAttempt, 'Peer not found');

    if AConfidence < 1 then
        raise EGSDAFailure.Create(fcProtocolFailure, 'Confidence 0 is not permitted');

    if AConfidence > 1000 then
        raise EGSDAFailure.Create(fcProtocolFailure, 'Confidence above 1000');

    LRunTask := FState.FindTask(ATaskID);
    if LRunTask = nil then
        raise EGSDAFailure.Create(fcCrossReferenceUnresolved, 'Task not found');

    if not VerifyPeerAuthentication(LPeer, AFingerprint, ARoundID, AToken) then
        raise EGSDAFailure.Create(fcImpersonationAttempt, 'Peer authentication failed');

    LPayload := ParsePayload(ARawPayload);
    try
        LPayloadSchema := LPayload.GetValue<string>('schema', '');
        if LPayloadSchema <> 'gsp/3.1' then
            raise EGSDAFailure.Create(fcSchemaFailure, 'Expected gsp/3.1 payload');

        LContentID := ComputeContentID(LPayload);
        if LPayload.GetValue<string>('content_id', '') <> LContentID then
            raise EGSDAFailure.Create(fcContentIdentityMismatch, 'content_id mismatch');

        LCanonical := TGSDAJsonCanonical.Canonicalize(LPayload);
        LSequence := FState.NextLogicalSequence(ARunID);

        LMessageID := TGSDAIdentityService.MessageID(
            ARunID,
            APeerID,
            LPayloadSchema,
            LContentID
        );

        if FState.FindMessage(LMessageID) <> nil then
            raise EGSDAFailure.Create(fcIdentityMismatch, 'Duplicate message identity');

        LMessage := TGSDAReasoningMessage.Create;
        LMessage.PeerID := APeerID;
        LMessage.RunID := ARunID;
        LMessage.TaskID := ATaskID;
        LMessage.RoundID := ARoundID;
        LMessage.LogicalSequence := LSequence;
        LMessage.Kind := AKind;
        LMessage.Confidence := AConfidence;
        LMessage.RawPayload := ARawPayload;
        LMessage.CanonicalPayload := LCanonical;
        LMessage.Payload := LPayload.Clone as TJSONObject;
        LMessage.ContentID := LContentID;
        LMessage.MessageID := LMessageID;

        LIdentityPreimage := TJSONObject.Create;
        try
            LIdentityPreimage.AddPair('run_id', ARunID);
            LIdentityPreimage.AddPair('peer_id', APeerID);
            LIdentityPreimage.AddPair('round_id', ARoundID);
            LIdentityPreimage.AddPair('content_id', LContentID);
            LIdentityPreimage.AddPair('payload_schema', LPayloadSchema);

            FState.AddIdentity(
                TGSDAIdentityRecord.Create(
                    LMessageID,
                    'message',
                    'message_identity/3.1',
                    TGSDAJsonCanonical.Canonicalize(LIdentityPreimage),
                    tzT4Kernel
                )
            );
        finally
            LIdentityPreimage.Free;
        end;

        FState.AddMessage(LMessage);

        Result.Accepted := True;
        Result.MessageID := LMessageID;
        Result.ContentID := LContentID;
        Result.ExecutionIdentity := LMessageID;
    finally
        LPayload.Free;
    end;
end;

function TGSDAKernel31.RegisterArtifact(

var
    LContentID: string;
    LPreimage: TJSONObject;

    if FState.FindTask(ATaskID) = nil then

    for var LParentID in AParentArtifactIDs do
        if FState.FindArtifact(LParentID) = nil then
            raise EGSDAFailure.Create(
                fcCrossReferenceUnresolved,
                'Parent artifact missing: ' + LParentID

    Result := TGSDAArtifact.Create;
    Result.TaskID := ATaskID;
    Result.RunID := ARunID;
    Result.PeerID := APeerID;
    Result.RoundID := ARoundID;
    Result.LogicalSequence := FState.NextLogicalSequence(ARunID);
    Result.Kind := AKind;
    Result.Content := AContent;
    Result.ContentID := LContentID;
    Result.ParentArtifactIDs := Copy(AParentArtifactIDs);
    Result.RequirementIDs := Copy(ARequirementIDs);
    Result.SourceRefs := Copy(ASourceRefs);
    Result.ArtifactID :=
        TGSDAIdentityService.ArtifactID(Result, AMessageID);

    FState.AddArtifact(Result);

    LPreimage := TJSONObject.Create;
    try
        LPreimage.AddPair('task_id', ATaskID);
        LPreimage.AddPair('run_id', ARunID);
        LPreimage.AddPair('artifact_kind', AKind);
        LPreimage.AddPair('content_id', LContentID);
        LPreimage.AddPair('author_peer_id', APeerID);

            CanonicalArrayOfIds(AParentArtifactIDs)

        FState.AddIdentity(

function TGSDAKernel31.ComputeCoverage(
    const ATaskID: string): TGSDACoverageResult;
var
    LTask: TGSDAResearchTask;
    LElement: TGSDAIntentElement;
    LArtifactID: string;
    LHasArtifact: Boolean;
    LUncovered: TList<Int64>;
begin
    LTask := FState.FindTask(ATaskID);
    if LTask = nil then
        raise EGSDAFailure.Create(fcCrossReferenceUnresolved, 'Task missing');

    FillChar(Result, SizeOf(Result), 0);
    LUncovered := TList<Int64>.Create;
    try
        for LElement in LTask.Intent.Elements do
        begin
            Inc(Result.TotalIntentElements);

            if LElement.Critical then
            begin
                Inc(Result.CriticalIntentElements);
                LHasArtifact := False;

                for LArtifactID in LElement.LinkedArtifactIDs do
                    if FState.FindArtifact(LArtifactID) <> nil then
                    begin
                        LHasArtifact := True;
                        Break;
                    end;

                if LHasArtifact then
                    Inc(Result.CoveredCriticalIntentElements)
                else
                    LUncovered.Add(LElement.ByteStart);
            end;
        end;

        if Result.CriticalIntentElements = 0 then
        begin
            Result.CriticalCoverageNumerator := 1;
            Result.CriticalCoverageDenominator := 1;
        end
        else
        begin
            Result.CriticalCoverageNumerator :=
                Result.CoveredCriticalIntentElements;
            Result.CriticalCoverageDenominator :=
                Result.CriticalIntentElements;
        end;

        Result.UncoveredOffsets := LUncovered.ToArray;
    finally
        LUncovered.Free;
    end;
end;

function TGSDAKernel31.ValidatePublication(
    const ASpecContentID: string;
    const ARequirementsJson, AGlossaryJson,
          AAssumptionsJson, AQuestionsJson,
          ATraceabilityJson: string;
    const AHumanDecision: TGSDAHumanDecision): Boolean;
var
    LExpectedID: string;
    LDecision: Boolean;
begin
    Result := False;

    LExpectedID := ComputeSpecContentID(
        ARequirementsJson,
        AGlossaryJson,
        AAssumptionsJson,
        AQuestionsJson,
        ATraceabilityJson
    );

    if LExpectedID <> ASpecContentID then
        Exit(False);

    if AHumanDecision = nil then
        Exit(False);

    if AHumanDecision.SpecContentID <> ASpecContentID then
        Exit(False);

    if not AHumanDecision.Accepted then
        Exit(False);

    if AHumanDecision.ActorID = '' then
        Exit(False);

    if AHumanDecision.Rationale = '' then
        Exit(False);

    if not IsHexString(AHumanDecision.SignatureHex) then
        Exit(False);

    if Length(AHumanDecision.SignatureHex) < 16 then
        Exit(False);

    for var LRequirement in FState.Requirements.Values do
        for var LSourceRef in LRequirement.SourceRefs do
            if FState.FindIdentity(LSourceRef) = nil then
                Exit(False);

    LDecision := AHumanDecision.Accepted;
    Result := LDecision;
end;

function TGSDAKernel31.CommitRevision(
    const ASpecContentID: string;
    const APackageFiles: TDictionary<string, string>;
    const ACanonicalManifest, APackageHash: string;
    const AHumanDecision: TGSDAHumanDecision): TGSDARevisionSnapshot;
var
    LPrevious: TGSDARevisionSnapshot;
begin
    if AHumanDecision = nil then
        raise EGSDAFailure.Create(fcGovernanceFailure, 'Human decision required');

    if not AHumanDecision.Accepted then
        raise EGSDAFailure.Create(fcGovernanceFailure, 'Human decision rejected');

    LPrevious := FState.RevisionStore.Latest;

    Result := TGSDARevisionSnapshot.Create;
    Result.SpecContentID := ASpecContentID;
    Result.PreviousRevisionID := '';
    Result.PackageHash := APackageHash;
    Result.CanonicalPackageManifest := ACanonicalManifest;
    Result.CreatedAtUtc := GSDA_UtcNowIso;

    if LPrevious <> nil then
        Result.PreviousRevisionID := LPrevious.RevisionID;

    Result.RevisionID :=
        TGSDAIdentityService.RevisionID(
            [Result.PreviousRevisionID],
            ASpecContentID,
            ACanonicalManifest
        );

    for var LPair in APackageFiles do
        Result.PackageFiles.Add(LPair.Key, LPair.Value);

    FState.RevisionStore.Add(Result);
end;

end.

procedure TGSDAPublication31.CheckNormativeKeywordLeaks(
    const ARequirementsJson, AGlossaryJson,
          AAssumptionsJson, AQuestionsJson,
          ATraceabilityJson: string);
const
    LKeywords: array[0..10] of string = (
        'MUST', 'MUST NOT', 'SHALL', 'SHALL NOT',
        'SHOULD', 'SHOULD NOT', 'MAY', 'REQUIRED',
        'RECOMMENDED', 'NOT RECOMMENDED', 'OPTIONAL'
    );
var
    LText: string;
    LKeyword: string;
begin
    LText :=
        UpperCase(
            ARequirementsJson + #10 +
            AGlossaryJson + #10 +
            AAssumptionsJson + #10 +
            AQuestionsJson + #10 +
            ATraceabilityJson
        );

    { Structured requirement data is allowed to carry the normative keyword
      in its dedicated field. This linter therefore only guards raw free-form
      serialized regions in this first implementation. }
    for LKeyword in LKeywords do
    begin
        if (Pos('"' + LKeyword + '"', LText) > 0) and
           (Pos('"normative_keyword":"' + LKeyword + '"', LText) = 0) then
            raise EGSDAFailure.Create(
                fcNormativeKeywordLeak,
                'Possible unmarked normative keyword: ' + LKeyword
            );
    end;
end;

procedure TGSDAPublication31.CheckNoCriticalFindings;
begin
    { No finding store exists in the authoritative kernel yet.
      Absence of a finding store is safe only because publication cannot
      synthesize a "pass"; this hook intentionally remains empty until
      a finding record is introduced. }
end;

procedure TGSDAPublication31.CheckTraceability;
var
    LRequirement: TGSDARequirement;
    LArtifactID: string;
    LSourceRef: string;
begin
    for LRequirement in FKernel.State.Requirements.Values do
    begin
        for LSourceRef in LRequirement.SourceRefs do
            if FKernel.State.FindIdentity(LSourceRef) = nil then
                raise EGSDAFailure.Create(
                    fcSourceReferenceUnresolved,
                    'Unresolved requirement source reference'
                );

        for LArtifactID in LRequirement.ArtifactIDs do
            if FKernel.State.FindArtifact(LArtifactID) = nil then
                raise EGSDAFailure.Create(
                    fcCrossReferenceUnresolved,
                    'Requirement artifact reference unresolved'
                );
    end;
end;

procedure TGSDAPublication31.CheckCoverage(const ATaskID: string);
var
    LCoverage: TGSDACoverageResult;
    LLeft, LRight: Int64;
begin
    LCoverage := FKernel.ComputeCoverage(ATaskID);

    LLeft :=
        LCoverage.CriticalCoverageNumerator *
        FKernel.Config.MinCriticalCoverageDenominator;

    LRight :=
        FKernel.Config.MinCriticalCoverageNumerator *
        LCoverage.CriticalCoverageDenominator;

    if LLeft < LRight then
        raise EGSDAFailure.Create(
            fcPublicationFailure,
            'Critical intent coverage threshold not satisfied'
        );
end;

procedure TGSDAPublication31.CheckReferences;
var
    LMessage: TGSDAReasoningMessage;
    LParentValue: TJSONValue;
    LParentID: string;
begin
    for LMessage in FKernel.State.Messages.Values do
    begin
        LParentValue := LMessage.Payload.GetValue('parent_ids');
        if LParentValue is TJSONArray then
            for var LI := 0 to TJSONArray(LParentValue).Count - 1 do
            begin
                LParentID := TJSONArray(LParentValue).Items[LI].Value;
                if (FKernel.State.FindMessage(LParentID) = nil) and
                   (FKernel.State.FindArtifact(LParentID) = nil) and
                   (FKernel.State.FindRequirement(LParentID) = nil) then
                    raise EGSDAFailure.Create(
                        fcCrossReferenceUnresolved,
                        'Unresolved parent reference'
                    );
            end;
    end;
end;

procedure TGSDAPublication31.CheckPackage(
    const APackageFiles: TDictionary<string, string>);
const
    LRequired: array[0..5] of string = (
        'spec.requirements.json',
        'spec.glossary.json',
        'spec.assumptions.json',
        'spec.open_questions.json',
        'spec.traceability.json',
        'registry.json'
    );
var
    LName: string;
begin
    for LName in LRequired do
        if not APackageFiles.ContainsKey(LName) then
            raise EGSDAFailure.Create(
                fcPublicationFailure,

    for LName in APackageFiles.Keys do

           LName.StartsWith('/') or
           LName.StartsWith('\') then
            raise EGSDAFailure.Create(
                fcPublicationFailure,
                'Unsafe package path: ' + LName
            );
end;

function TGSDAPublication31.ComputePackageHash(
    const APackageFiles: TDictionary<string, string>): string;
var
    LNames: TList<string>;
    LName: string;
    LManifest: TStringBuilder;
    LBytes: TBytes;
begin
    LNames := TList<string>.Create;
    try
        for LName in APackageFiles.Keys do
            LNames.Add(LName);

        LNames.Sort;

        LManifest := TStringBuilder.Create;
        try
            for LName in LNames do
            begin
                LBytes := TEncoding.UTF8.GetBytes(APackageFiles[LName]);
                LManifest.Append(LName);
                LManifest.Append(':');
                LManifest.Append(TGSDAIdentityService.FileContentID(LBytes));
                LManifest.Append(#10);
            end;

            Result := GSDA_SHA256Hex(LManifest.ToString);
        finally
            LManifest.Free;
        end;
    finally
        LNames.Free;
    end;
end;

function TGSDAPublication31.Publish(
    const ATaskID: string;
    const ARequirementsJson, AGlossaryJson,
          AAssumptionsJson, AQuestionsJson,
          ATraceabilityJson: string;
    const AHumanDecision: TGSDAHumanDecision): TGSDARevisionSnapshot;
var
    LSpecContentID: string;
    LPackage: TDictionary<string, string>;
    LPackageHash: string;
    LManifest: TStringBuilder;
    LNames: TList<string>;
    LName: string;
    LHumanDecision: TGSDAHumanDecision;
begin
    LSpecContentID := FKernel.ComputeSpecContentID(
        ARequirementsJson,
        AGlossaryJson,
        AAssumptionsJson,
        AQuestionsJson,
        ATraceabilityJson
    );

    CheckNormativeKeywordLeaks(
        ARequirementsJson,
        AGlossaryJson,
        AAssumptionsJson,
        AQuestionsJson,
        ATraceabilityJson
    );

    CheckCoverage(ATaskID);
    CheckTraceability;
    CheckReferences;
    CheckNoCriticalFindings;

       (not AHumanDecision.Accepted) or

            'Authenticated human acceptance required for exact SpecContentID'

    LHumanDecision := AHumanDecision;

    if not FKernel.ValidatePublication(
        LSpecContentID,
        ARequirementsJson,
        AGlossaryJson,
        AAssumptionsJson,
        AQuestionsJson,
        ATraceabilityJson,
        LHumanDecision
    ) then
        raise EGSDAFailure.Create(
            fcPublicationFailure,
            'Publication predicate failed'
        );

    LPackage := BuildPackageFiles(
        ARequirementsJson,
        AGlossaryJson,
        AAssumptionsJson,
        AQuestionsJson,
        ATraceabilityJson
    );
    try
        LPackageHash := ComputePackageHash(LPackage);

        LNames := TList<string>.Create;
        try
            for LName in LPackage.Keys do
                LNames.Add(LName);
            LNames.Sort;

            LManifest := TStringBuilder.Create;
            try
                for LName in LNames do
                    LManifest.Append(LName)
                        .Append('=')
                        .Append(TGSDAIdentityService.FileContentID(
                            TEncoding.UTF8.GetBytes(LPackage[LName])))
                        .Append(#10);

                Result := FKernel.CommitRevision(
                    LSpecContentID,
                    LPackage,
                    LManifest.ToString,
                    LPackageHash,
                    LHumanDecision
                );
            finally
                LManifest.Free;
            end;
        finally
            LNames.Free;
        end;
    finally
        LPackage.Free;
    end;
end;

end.
'''
