GRISP Delphi 13 Core
====================

This bundle contains a minimal, deterministic GRISP core implemented in Delphi 13.

Files:
- src/GrispCLI.dpr        : CLI harness
- src/GrispLexer.pas      : deterministic lexer
- src/GrispParser.pas     : surface-language parser
- src/GrispExecutor.pas   : deterministic executor
- docs/grammar.grisp      : surface grammar

Build:
  dcc64 src/GrispCLI.dpr

Run:
  GrispCLI.exe < program.grisp

Notes:
- No FPC syntax.
- No {mode} directives.
- CharInSet used for Unicode-safe character classification.
- No examples; only implementation.

# GRISP Surface Grammar (Delphi 13 Core)

program        ::= rules_block

rules_block    ::= "rules" "begin" { rule_decl } "end"

rule_decl      ::= "rule" STRING ["priority" INT] ["scale" INT] ["fairness" INT] "begin"
                   match_block [where_block] [let_block] actions_block "end"

match_block    ::= "match" "begin" { pattern } "end"
pattern        ::= IDENT ":" TYPE_NAME ";"

where_block    ::= "where" EXPR ";"

let_block      ::= "let" "begin" { let_stmt } "end"
let_stmt       ::= IDENT ":=" EXPR ";"

actions_block  ::= "actions" "begin" { action_stmt } "end"
action_stmt    ::= CreateNodeStmt | UpdateFieldStmt | DeleteNodeStmt | EmitEventStmt | QueryStmt ";"

CreateNodeStmt ::= "CreateNode" "(" IDENT "," TYPE_NAME ")" "with" "{" map_literal "}" ";"
UpdateFieldStmt::= "UpdateField" "(" IDENT "," IDENT "," EXPR ")" ";"
DeleteNodeStmt ::= "DeleteNode" "(" IDENT ")" ";"
EmitEventStmt  ::= "EmitEvent" "(" STRING "," list_expr ")" ";"
QueryStmt      ::= "Query" "(" STRING ")" ";"

map_literal    ::= kv_pair { ";" kv_pair } [ ";" ]
kv_pair        ::= IDENT ":" EXPR

list_expr      ::= "[" [ EXPR { "," EXPR } ] "] "

EXPR           ::= literal | IDENT | IDENT "." IDENT | EXPR BINOP EXPR | "len(" EXPR ")"
BINOP          ::= "+" | "-" | "*" | "/" | "==" | "!=" | "<" | "<=" | ">" | ">=" | "&&" | "||"

unit GrispLexer;

interface

uses
  SysUtils;

type
  TTokenKind = (
    tkEOF, tkIdent, tkString, tkNumber,
    tkLParen, tkRParen, tkLBrace, tkRBrace,
    tkLBracket, tkRBracket, tkComma, tkColon,
    tkSemicolon, tkDot, tkOp, tkKeyword
  );

  TToken = record
    Kind: TTokenKind;
    Text: string;
    Line: Integer;
    Col: Integer;
  end;

  TLexer = class
  private
    FText: string;
    FPos: Integer;
    FLine: Integer;
    FCol: Integer;
    function Peek: Char;
    function NextChar: Char;
    procedure SkipWhitespaceAndComments;
    function ReadIdentOrKeyword: TToken;
    function ReadNumber: TToken;
    function ReadString: TToken;
  public
    constructor Create(const AText: string);
    function NextToken: TToken;
  end;

implementation

constructor TLexer.Create(const AText: string);
begin
  FText := AText;
  FPos := 1;
  FLine := 1;
  FCol := 1;
end;

function TLexer.Peek: Char;
begin
  if FPos > Length(FText) then Exit(#0);
  Result := FText[FPos];
end;

function TLexer.NextChar: Char;
begin
  if FPos > Length(FText) then Exit(#0);
  Result := FText[FPos];
  Inc(FPos);
  if Result = #10 then
  begin
    Inc(FLine);
    FCol := 1;
  end
  else
    Inc(FCol);
end;

procedure TLexer.SkipWhitespaceAndComments;
begin
  while True do
  begin
    if CharInSet(Peek, [' ', #9, #10, #13]) then
    begin
      NextChar;
      Continue;
    end;

    if (Peek = '/') and (FPos < Length(FText)) and (FText[FPos+1] = '/') then
    begin
      while (Peek <> #0) and (Peek <> #10) do NextChar;
      Continue;
    end;

    if (Peek = '/') and (FPos < Length(FText)) and (FText[FPos+1] = '*') then
    begin
      NextChar; NextChar;
      while not ((Peek = '*') and (FPos < Length(FText)) and (FText[FPos+1] = '/')) do
      begin
        if Peek = #0 then Exit;
        NextChar;
      end;
      NextChar; NextChar;
      Continue;
    end;

    Break;
  end;
end;

function TLexer.ReadIdentOrKeyword: TToken;
var
  sb: string;
begin
  sb := '';
  while CharInSet(Peek, ['A'..'Z','a'..'z','0'..'9','_']) do
    sb := sb + NextChar;

  Result.Kind := tkIdent;
  Result.Text := sb;
  Result.Line := FLine;
  Result.Col := FCol;

  if SameText(sb, 'rule') or SameText(sb, 'match') or SameText(sb, 'actions') or
     SameText(sb, 'begin') or SameText(sb, 'end') or SameText(sb, 'where') or
     SameText(sb, 'let') or SameText(sb, 'CreateNode') or SameText(sb, 'UpdateField') or
     SameText(sb, 'DeleteNode') or SameText(sb, 'EmitEvent') or SameText(sb, 'Query') then
    Result.Kind := tkKeyword;
end;

function TLexer.ReadNumber: TToken;
var
  sb: string;
begin
  sb := '';
  while CharInSet(Peek, ['0'..'9']) do
    sb := sb + NextChar;

  Result.Kind := tkNumber;
  Result.Text := sb;
  Result.Line := FLine;
  Result.Col := FCol;
end;

function TLexer.ReadString: TToken;
var
  sb: string;
  c: Char;
begin
  sb := '';
  NextChar;
  while True do
  begin
    c := Peek;
    if c = #0 then Break;
    if c = '"' then
    begin
      NextChar;
      Break;
    end;
    if c = '\' then
    begin
      NextChar;
      case Peek of
        'n': begin sb := sb + #10; NextChar; end;
        'r': begin sb := sb + #13; NextChar; end;
        't': begin sb := sb + #9; NextChar; end;
        '"': begin sb := sb + '"'; NextChar; end;
        '\': begin sb := sb + '\'; NextChar; end;
      else
        sb := sb + NextChar;
      end;
    end
    else
      sb := sb + NextChar;
  end;

  Result.Kind := tkString;
  Result.Text := sb;
  Result.Line := FLine;
  Result.Col := FCol;
end;

function TLexer.NextToken: TToken;
var
  c: Char;
begin
  SkipWhitespaceAndComments;
  c := Peek;

  Result.Line := FLine;
  Result.Col := FCol;

  if c = #0 then
  begin
    Result.Kind := tkEOF;
    Result.Text := '';
    Exit;
  end;

  if CharInSet(c, ['A'..'Z','a'..'z','_']) then Exit(ReadIdentOrKeyword);
  if CharInSet(c, ['0'..'9']) then Exit(ReadNumber);

  case c of
    '(' : begin NextChar; Result.Kind := tkLParen; Result.Text := '('; end;
    ')' : begin NextChar; Result.Kind := tkRParen; Result.Text := ')'; end;
    '{' : begin NextChar; Result.Kind := tkLBrace; Result.Text := '{'; end;
    '}' : begin NextChar; Result.Kind := tkRBrace; Result.Text := '}'; end;
    '[' : begin NextChar; Result.Kind := tkLBracket; Result.Text := '['; end;
    ']' : begin NextChar; Result.Kind := tkRBracket; Result.Text := ']'; end;
    ',' : begin NextChar; Result.Kind := tkComma; Result.Text := ','; end;
    ':' : begin NextChar; Result.Kind := tkColon; Result.Text := ':'; end;
    ';' : begin NextChar; Result.Kind := tkSemicolon; Result.Text := ';'; end;
    '.' : begin NextChar; Result.Kind := tkDot; Result.Text := '.'; end;
    '"' : Exit(ReadString);
  else
    NextChar;
    Result.Kind := tkOp;
    Result.Text := c;
  end;
end;

end.

unit GrispParser;

interface

uses
  SysUtils, Classes, GrispLexer;

type
  TActionKind = (akCreateNode, akUpdateField, akDeleteNode, akEmitEvent, akQuery);

  TAction = class
  public
    Kind: TActionKind;
    Name: string;
    AType: string;
    Params: TStringList;
    constructor Create;
    destructor Destroy; override;
  end;

  TRule = class
  public
    RuleId: string;
    Actions: TObjectList;
    constructor Create;
    destructor Destroy; override;
  end;

  TParser = class
  private
    FLexer: TLexer;
    FLook: TToken;
    procedure Next;
    procedure ExpectKind(k: TTokenKind);
    function ParseProgram: TObjectList;
    function ParseRule: TRule;
    procedure ParseMatch;
    procedure ParseWhere;
    function ParseActions: TObjectList;
    function ParseActionStmt: TAction;
  public
    constructor Create(const AText: string);
    destructor Destroy; override;
    function Parse: TObjectList;
  end;

implementation

constructor TAction.Create;
begin
  Params := TStringList.Create;
end;

destructor TAction.Destroy;
begin
  Params.Free;
  inherited Destroy;
end;

constructor TRule.Create;
begin
  Actions := TObjectList.Create(True);
end;

destructor TRule.Destroy;
begin
  Actions.Free;
  inherited Destroy;
end;

constructor TParser.Create(const AText: string);
begin
  FLexer := TLexer.Create(AText);
  FLook := FLexer.NextToken;
end;

destructor TParser.Destroy;
begin
  FLexer.Free;
  inherited Destroy;
end;

procedure TParser.Next;
begin
  FLook := FLexer.NextToken;
end;

procedure TParser.ExpectKind(k: TTokenKind);
begin
  if FLook.Kind <> k then
    raise Exception.CreateFmt('Parse error at line %d col %d', [FLook.Line, FLook.Col]);
  Next;
end;

function TParser.Parse: TObjectList;
begin
  Result := ParseProgram;
end;

function TParser.ParseProgram: TObjectList;
var
  rules: TObjectList;
begin
  rules := TObjectList.Create(True);

  if (FLook.Kind = tkKeyword) and SameText(FLook.Text, 'rules') then
  begin
    Next;
    ExpectKind(tkKeyword); // begin
    while not ((FLook.Kind = tkKeyword) and SameText(FLook.Text, 'end')) do
      rules.Add(ParseRule);
    ExpectKind(tkKeyword); // end
  end
  else
    raise Exception.Create('Program must start with rules begin ... end');

  Result := rules;
end;

function TParser.ParseRule: TRule;
var
  r: TRule;
begin
  r := TRule.Create;

  if not ((FLook.Kind = tkKeyword) and SameText(FLook.Text, 'rule')) then
    raise Exception.Create('Expected rule');

  Next;

  if FLook.Kind <> tkString then
    raise Exception.Create('Expected rule id string');

  r.RuleId := FLook.Text;
  Next;

  while not ((FLook.Kind = tkKeyword) and SameText(FLook.Text, 'begin')) do
    Next;

  ExpectKind(tkKeyword); // begin

  if (FLook.Kind = tkKeyword) and SameText(FLook.Text, 'match') then
    ParseMatch;

  if (FLook.Kind = tkKeyword) and SameText(FLook.Text, 'where') then
    ParseWhere;

  if (FLook.Kind = tkKeyword) and SameText(FLook.Text, 'let') then
  begin
    Next;
    ExpectKind(tkKeyword); // begin
    while not ((FLook.Kind = tkKeyword) and SameText(FLook.Text, 'end')) do
      Next;
    ExpectKind(tkKeyword); // end
  end;

  r.Actions.Assign(ParseActions);

  if (FLook.Kind = tkKeyword) and SameText(FLook.Text, 'end') then
    Next
  else
    raise Exception.Create('Expected end of rule');

  Result := r;
end;

procedure TParser.ParseMatch;
begin
  ExpectKind(tkKeyword); // match
  ExpectKind(tkKeyword); // begin
  while not ((FLook.Kind = tkKeyword) and SameText(FLook.Text, 'end')) do
    Next;
  ExpectKind(tkKeyword); // end
end;

procedure TParser.ParseWhere;
begin
  ExpectKind(tkKeyword); // where
  while FLook.Kind <> tkSemicolon do Next;
  ExpectKind(tkSemicolon);
end;

function TParser.ParseActions: TObjectList;
var
  list: TObjectList;
begin
  list := TObjectList.Create(True);

  ExpectKind(tkKeyword); // actions
  ExpectKind(tkKeyword); // begin

  while not ((FLook.Kind = tkKeyword) and SameText(FLook.Text, 'end')) do
    list.Add(ParseActionStmt);

  ExpectKind(tkKeyword); // end

  Result := list;
end;

function TParser.ParseActionStmt: TAction;
var
  act: TAction;
begin
  act := TAction.Create;

  if (FLook.Kind = tkKeyword) and SameText(FLook.Text, 'CreateNode') then
  begin
    Next;
    ExpectKind(tkLParen);
    if FLook.Kind <> tkIdent then raise Exception.Create('Expected ident');
    act.Name := FLook.Text; Next;
    ExpectKind(tkComma);
    if FLook.Kind <> tkIdent then raise Exception.Create('Expected type');
    act.AType := FLook.Text; Next;
    ExpectKind(tkRParen);

    if (FLook.Kind = tkKeyword) and SameText(FLook.Text, 'with') then
    begin
      Next;
      ExpectKind(tkLBrace);
      while FLook.Kind <> tkRBrace do Next;
      ExpectKind(tkRBrace);
    end;

    ExpectKind(tkSemicolon);
    act.Kind := akCreateNode;
  end
  else if (FLook.Kind = tkKeyword) and SameText(FLook.Text, 'UpdateField') then
  begin
    Next;
    ExpectKind(tkLParen);
    if FLook.Kind <> tkIdent then raise Exception.Create('Expected ident');
    act.Name := FLook.Text; Next;
    ExpectKind(tkComma);
    if FLook.Kind <> tkIdent then raise Exception.Create('Expected field');
    act.Params.Add(FLook.Text); Next;
    ExpectKind(tkComma);
    while FLook.Kind <> tkRParen do Next;
    ExpectKind(tkRParen);
    ExpectKind(tkSemicolon);
    act.Kind := akUpdateField;
  end
  else if (FLook.Kind = tkKeyword) and SameText(FLook.Text, 'DeleteNode') then
  begin
    Next;
    ExpectKind(tkLParen);
    if FLook.Kind <> tkIdent then raise Exception.Create('Expected ident');
    act.Name := FLook.Text; Next;
    ExpectKind(tkRParen);
    ExpectKind(tkSemicolon);
    act.Kind := akDeleteNode;
  end
  else if (FLook.Kind = tkKeyword) and SameText(FLook.Text, 'EmitEvent') then
  begin
    Next;
    ExpectKind(tkLParen);
    if FLook.Kind <> tkString then raise Exception.Create('Expected event type');
    act.AType := FLook.Text; Next;
    ExpectKind(tkComma);
    // parse list expression minimally: "[" ... "]"
    ExpectKind(tkLBracket);
    while FLook.Kind <> tkRBracket do
      Next;
    ExpectKind(tkRBracket);
    ExpectKind(tkRParen);
    ExpectKind(tkSemicolon);
    act.Kind := akEmitEvent;
  end
  else if (FLook.Kind = tkKeyword) and SameText(FLook.Text, 'Query') then
  begin
    Next;
    ExpectKind(tkLParen);
    if FLook.Kind <> tkString then
      raise Exception.Create('Expected query string');
    act.AType := FLook.Text;
    Next;
    ExpectKind(tkRParen);
    ExpectKind(tkSemicolon);
    act.Kind := akQuery;
  end
  else
    raise Exception.CreateFmt('Unknown action at line %d col %d', [FLook.Line, FLook.Col]);

  Result := act;
end;

end.
