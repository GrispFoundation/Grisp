unit Grisp.Types;

interface
uses
  System.SysUtils, System.Generics.Collections;

type
  // Identifier type: TypeName and sequence number
  TID = record
    TypeName: string;
    SeqNum: Int64;
  end;

  // Value kinds follow the GRISP canonical order tags
  TValueKind = (vkInt, vkFixed, vkBool, vkString, vkIdent, vkList, vkMap);

  // Fixed point representation
  TFixed = record
    Scale: Integer;
    Scaled: Int64; // scaled integer representation
  end;

  // Canonical value representation (compact)
  TValue = record
    Kind: TValueKind;
    IntVal: Int64;
    FixedVal: TFixed;
    BoolVal: Boolean;
    StrVal: string;
    IdentVal: TID;
    ListVal: TArray<TValue>;
    MapVal: TDictionary<TValue, TValue>;
  end;

  // Field map for nodes/edges
  TFieldMap = TDictionary<string, TValue>;

  // Node record
  TNode = record
    Id: TID;
    TypeName: string;
    Fields: TFieldMap;
  end;

  // Edge record
  TEdge = record
    Id: TID;
    TypeName: string;
    Source, Target: TID;
    Fields: TFieldMap;
  end;

  // Graph container
  TGraph = record
    Nodes: TDictionary<TID, TNode>;
    Edges: TDictionary<TID, TEdge>;
  end;

implementation

end.

unit Grisp.Order;

interface
uses
  System.SysUtils, System.Generics.Collections, Grisp.Types;

type
  // Compare result: -1,0,1
  TCompareResult = -1..1;

function CompareTID(const A, B: TID): Integer;
function CompareValue(const A, B: TValue): Integer;

implementation

function CompareTID(const A, B: TID): Integer;
begin
  Result := CompareText(A.TypeName, B.TypeName);
  if Result = 0 then
  begin
    if A.SeqNum < B.SeqNum then Exit(-1);
    if A.SeqNum > B.SeqNum then Exit(1);
    Exit(0);
  end;
  if Result < 0 then Exit(-1) else Exit(1);
end;

function CompareValue(const A, B: TValue): Integer;
var
  i, n: Integer;
  ka, kb: Integer;
begin
  // Type tag order: Integer < FixedPoint < Boolean < String < Identifier < List < Map
  ka := Ord(A.Kind); kb := Ord(B.Kind);
  if ka < kb then Exit(-1);
  if ka > kb then Exit(1);

  case A.Kind of
    vkInt:
      if A.IntVal < B.IntVal then Exit(-1) else if A.IntVal > B.IntVal then Exit(1) else Exit(0);
    vkFixed:
      begin
        if A.FixedVal.Scale < B.FixedVal.Scale then Exit(-1)
        else if A.FixedVal.Scale > B.FixedVal.Scale then Exit(1)
        else if A.FixedVal.Scaled < B.FixedVal.Scaled then Exit(-1)
        else if A.FixedVal.Scaled > B.FixedVal.Scaled then Exit(1)
        else Exit(0);
      end;
    vkBool:
      if A.BoolVal = B.BoolVal then Exit(0) else if not A.BoolVal and B.BoolVal then Exit(-1) else Exit(1);
    vkString:
      begin
        if A.StrVal < B.StrVal then Exit(-1) else if A.StrVal > B.StrVal then Exit(1) else Exit(0);
      end;
    vkIdent:
      Exit(CompareTID(A.IdentVal, B.IdentVal));
    vkList:
      begin
        n := Min(Length(A.ListVal), Length(B.ListVal));
        for i := 0 to n - 1 do
        begin
          Result := CompareValue(A.ListVal[i], B.ListVal[i]);
          if Result <> 0 then Exit;
        end;
        if Length(A.ListVal) < Length(B.ListVal) then Exit(-1)
        else if Length(A.ListVal) > Length(B.ListVal) then Exit(1)
        else Exit(0);
      end;
    vkMap:
      begin
        // Map comparison must follow ORDER_CANONICAL: compare tuples of (order_key(k), order_key(v))
        // This implementation provides a deterministic placeholder: compare counts then stringified keys.
        if A.MapVal.Count < B.MapVal.Count then Exit(-1)
        else if A.MapVal.Count > B.MapVal.Count then Exit(1);
        var keysA := TList<string>.Create;
        try
          for var kv in A.MapVal.Keys do keysA.Add(kv.StrVal);
          keysA.Sort;
          var keysB := TList<string>.Create;
          try
            for var kv in B.MapVal.Keys do keysB.Add(kv.StrVal);
            keysB.Sort;
            for i := 0 to keysA.Count - 1 do
            begin
              if keysA[i] < keysB[i] then Exit(-1) else if keysA[i] > keysB[i] then Exit(1);
              // Value comparison by key is omitted here for brevity; full implementation must compare order_key(v)
            end;
          finally keysB.Free; end;
        finally keysA.Free; end;
        Exit(0);
      end;
  else
    Exit(0);
  end;
end;

end.
