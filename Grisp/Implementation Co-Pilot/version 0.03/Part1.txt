unit Grisp.Types;

interface
uses
  System.SysUtils, System.Generics.Collections;

type
  // Identifier type: TypeName and sequence number
  TID = record
    TypeName: string;
    SeqNum: Int64;
  end;

  // Value kinds follow the GRISP canonical order tags
  TValueKind = (vkInt, vkFixed, vkBool, vkString, vkIdent, vkList, vkMap);

  // Fixed point representation
  TFixed = record
    Scale: Integer;
    Scaled: Int64; // scaled integer representation
  end;

  // Canonical value representation (compact)
  TValue = record
    Kind: TValueKind;
    IntVal: Int64;
    FixedVal: TFixed;
    BoolVal: Boolean;
    StrVal: string;
    IdentVal: TID;
    ListVal: TArray<TValue>;
    MapVal: TDictionary<TValue, TValue>;
  end;

  // Field map for nodes/edges
  TFieldMap = TDictionary<string, TValue>;

  // Node record
  TNode = record
    Id: TID;
    TypeName: string;
    Fields: TFieldMap;
  end;

  // Edge record
  TEdge = record
    Id: TID;
    TypeName: string;
    Source, Target: TID;
    Fields: TFieldMap;
  end;

  // Graph container
  TGraph = record
    Nodes: TDictionary<TID, TNode>;
    Edges: TDictionary<TID, TEdge>;
  end;

implementation

end.
