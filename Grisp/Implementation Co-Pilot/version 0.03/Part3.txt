unit Grisp.Engine;

interface
uses
  Grisp.Types, Grisp.Counters, Grisp.Match, System.Generics.Collections;

type
  TConfig = record
    G: TGraph;
    C: TCounters;
    MPrev: TDictionary<string, Int64>;
    Rejected: TDictionary<string, Boolean>;
  end;

procedure InitConfig(var CFG: TConfig);
procedure FreeConfig(var CFG: TConfig);
procedure Delta(var CFG: TConfig; const IR: TArray<string>);

implementation
uses
  Grisp.Graph, Grisp.Compiler, System.SysUtils;

procedure InitConfig(var CFG: TConfig);
begin
  InitGraph(CFG.G);
  InitCounters(CFG.C);
  CFG.MPrev := TDictionary<string, Int64>.Create;
  CFG.Rejected := TDictionary<string, Boolean>.Create;
end;

procedure FreeConfig(var CFG: TConfig);
begin
  FreeGraph(CFG.G);
  FreeCounters(CFG.C);
  CFG.MPrev.Free;
  CFG.Rejected.Free;
end;

procedure Delta(var CFG: TConfig; const IR: TArray<string>);
var
  Matches: TList<TMatch>;

  Alpha: TAlpha;
  ReadSet: TList<TReadEntry>;
begin
  // DISCOVER: EvalPatterns must enumerate matches deterministically
  Matches := TList<TMatch>.Create;
  try
    // If no matches, NO_MATCH: advance tick counter

      CFG.C.TickCounter := CFG.C.TickCounter + 1;
      Exit;
    end;

    // SELECT: choose highest score then lexicographic match_key; skip Rejected
