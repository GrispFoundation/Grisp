; GRISP V1.2 Core Grammar (syntax only)

<program> ::= { <declaration> }

<declaration> ::= <node_decl>
                | <edge_decl>
                | <function_decl>
                | <rule_decl>

<node_decl> ::= "node" <identifier> "{" { <field_decl> } "}"
<field_decl> ::= <identifier> ":" <type> [ "?" ]

<edge_decl> ::= "edge" <identifier> "(" <type> "->" <type> ")"

<type> ::= "identifier" | "string" | "number" | "integer" | "boolean"
         | "array" "<" <type> ">"
         | <enum_type>
         | <identifier>

<enum_type> ::= "enum" "(" <string_list> ")"

; Functions: signatures and optional bodies or external marker
<pure_fn_decl> ::= "pure" "fn" <identifier> "(" [ <param_list> ] ")" ":" <type> [ <fn_body_or_external> ]
<effect_fn_decl> ::= "effect" "(" <effect_meta> ")" "fn" <identifier> "(" [ <param_list> ] ")" ":" <type> [ <fn_body_or_external> ]
<nondet_fn_decl> ::= "nondeterministic" "fn" <identifier> "(" [ <param_list> ] ")" ":" <type> [ <fn_body_or_external> ]

<fn_body_or_external> ::= "external" | <fn_body>
<fn_body> ::= "{" { <statement> } "}"  ; optional small expression language for inline functions

<effect_meta> ::= "idempotent" "=" ("true" | "false") "," "timeout_ms" "=" <integer> "," "on_timeout" "=" <on_timeout_val>
<on_timeout_val> ::= "retry" | "fail" | "return_default" | "propagate" | "escalate"

<param_list> ::= <param> { "," <param> }
<param> ::= <identifier> ":" <type>

; Rules
<rule_decl> ::= "rule" <identifier> "priority" "=" <integer> "{" <rule_body> "}"
<rule_body> ::= <metadata_block> <when_block> <then_block> [ <hooks_block> ]

<metadata_block> ::= "metadata" "{" { <meta_field> } "}"
<meta_field> ::= "idempotent" "=" ("true" | "false")
               | "depends_on" "=" "[" <id_list> "]"
               | "emits" "=" <array_literal>

; WHEN block: patterns and bindings
<when_block> ::= "when" "{" { <pattern_line> } "}"
<pattern_line> ::= <binding>
                 | <let_binding>
                 | <collection_binding>
                 | <existence_expr>
                 | <condition_expr>
                 | <edge_pattern>

<binding> ::= <var_decl> "where" <predicate>
<var_decl> ::= <identifier> ":" <type>  ; aliasing: A: Agent where ...
<let_binding> ::= "let" <identifier> "=" <pure_expr>
<collection_binding> ::= <identifier> "many" <type> "where" <predicate>
<existence_expr> ::= "exists" <pattern> | "not" "exists" <pattern>
<condition_expr> ::= "condition" <pure_expr>

<edge_pattern> ::= "edge" <identifier> "(" <pattern_ref> "->" <pattern_ref> ")"
<pattern> ::= <var_decl> "where" <predicate> | <edge_pattern>
<pattern_ref> ::= <identifier> | <identifier> "." <field_name>

; THEN block: actions
<then_block> ::= "then" "{" { <action> } "}"
<action> ::= <let_action> | <update_action> | <create_action> | <delete_action> | <emit_action> | <for_action> | <call_action>

<let_action> ::= "let" <identifier> "=" <expr>
<update_action> ::= "update" <identifier> "{" { <assign> } "}"
<create_action> ::= "create" <create_target>
<delete_action> ::= "delete" <create_target> [ "when" <expr> ]
<emit_action> ::= "emit" "Event" "(" <emit_args> ")"
<for_action> ::= "for" <identifier> "in" <collection> "{" { <action> } "}"
<call_action> ::= "call" <identifier> "(" [ <arg_list> ] ")"

<create_target> ::= <type> "{" { <assign> } "}"
                  | <identifier> ":" <type> "{" { <assign> } "}"
                  | <edge_create>

<edge_create> ::= <identifier> "(" <identifier> "->" <identifier> ")"
<assign> ::= <field_name> ":" <expr>

<emit_args> ::= "type" "=" <string> "," "class" "=" <event_class> [ "," "payload" "=" <expr> ] [ "," "emit_hint" "=" <emit_hint> ]
<event_class> ::= "audit" | "operational" | "debug"
<emit_hint> ::= "{" "sample_rate" ":" <number> [ "," "aggregate_key" ":" <string> ] "}"

<hooks_block> ::= "hooks" "{" { <hook_entry> } "}"
<hook_entry> ::= "after_commit" ":" <array_literal> | "on_failure" ":" <array_literal>

; Expressions with precedence (parser-friendly)
<expr> ::= <or_expr>

<or_expr> ::= <and_expr> { "or" <and_expr> }
<and_expr> ::= <cmp_expr> { "and" <cmp_expr> }
<cmp_expr> ::= <add_expr> [ ( "==" | "!=" | "<" | ">" | "<=" | ">=" ) <add_expr> ]
<add_expr> ::= <mul_expr> { ( "+" | "-" ) <mul_expr> }
<mul_expr> ::= <unary_expr> { ( "*" | "/" ) <unary_expr> }
<unary_expr> ::= [ "-" | "not" ] <primary_expr>
<primary_expr> ::= <literal> | <identifier> | <identifier> "." <field_name> | <fn_call> | "(" <expr> ")"

<fn_call> ::= <identifier> "(" [ <arg_list> ] ")"
<arg_list> ::= <expr> { "," <expr> }

<object_literal> ::= "{" [ <object_field> { "," <object_field> } ] "}"
<object_field> ::= <identifier> ":" <expr>
<array_literal> ::= "[" [ <expr> { "," <expr> } ] "]"

<number> ::= <integer> | <float>
<integer> ::= [0-9]+
<float> ::= [0-9]+ "." [0-9]+

<id_list> ::= <identifier> { "," <identifier> }
<string_list> ::= <string> { "," <string> }

<identifier> ::= [A-Za-z_][A-Za-z0-9_]*
<string> ::= '"' [^"]* '"'
