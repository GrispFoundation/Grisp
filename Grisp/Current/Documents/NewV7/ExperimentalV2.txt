# Canonicalization: UTF-8 bytewise key sorting, no extra whitespace

def canonicalize_json(obj: Any) -> str:
    """Convert any JSON-compatible object to GRISP Canonical JSON string."""
    if isinstance(obj, dict):
        items = sorted(obj.items(), key=lambda kv: kv[0].encode('utf-8'))
        return '{' + ','.join(f'{json.dumps(k, ensure_ascii=False)}:{canonicalize_json(v)}' for k, v in items) + '}'
    if isinstance(obj, list):
        return '[' + ','.join(canonicalize_json(e) for e in obj) + ']'
    return json.dumps(obj, ensure_ascii=False)

def strip_non_normative(obj: Any) -> Any:
    """Remove non-normative metadata fields (e.g., 'metadata', 'timestamp')."""
    if isinstance(obj, dict):
        return {k: strip_non_normative(v) for k, v in obj.items() if k not in ('metadata', 'diagnostic', 'timestamp', 'engine_version')}
    if isinstance(obj, list):

            errors.append(f"adjacency_version[Person:0][Friendship][outgoing] = {p0_adj}, expected 1")
        p1_adj = adj.get('Person:1', {}).get('Friendship', {}).get('incoming')
        if p1_adj != 1:
            errors.append(f"adjacency_version[Person:1][Friendship][incoming] = {p1_adj}, expected 1")
        # ordering dependency for name
        odv = vc.get('ordering_dependency_version', {}).get('Person', {}).get('name')
        if odv != 1:
            errors.append(f"ordering_dependency_version.Person.name = {odv}, expected 1")

    # Test Vector 3 (no-op) – no extra counters beyond creation
    if manifest_id == 'test-basic':
        # already covered
        pass

    # Test Vector 4 (across ticks) – handled by aging manifest
    if manifest_id == 'test-aging':
        p = nodes.get('Person:0')
        if p and p.get('field_versions', {}).get('age') != 3:
            errors.append(f"Person:0.age field_version = {p.get('field_versions', {}).get('age')}, expected 3")
        if p.get('existence_version') != 1:
            errors.append("existence_version changed across ticks; should remain 1")
        ev = vc.get('extent_version', {}).get('Person')
        if ev != 1:
            errors.append(f"extent_version.Person = {ev}, expected 1 (unchanged across updates)")

    # Test Vector 5 (adjacency increments on node deletion)
    if manifest_id == 'test-delete':
        et = vc.get('edge_type_version', {}).get('Friendship')
        if et != 2:
            errors.append(f"edge_type_version.Friendship = {et}, expected 2 (cumulative create+delete)")
        adj = vc.get('adjacency_version', {}).get('Person:1', {}).get('Friendship', {}).get('incoming')
        if adj != 2:
            errors.append(f"adjacency_version[Person:1][Friendship][incoming] = {adj}, expected 2")
        # Person:0 should be gone
        if 'Person:0' in nodes:
            errors.append("Person:0 still present after deletion")
        # Edge should be gone
        if edges:
            errors.append(f"Edges still present: {edges}")

    # Test Vector 6 (ordering dependency) – no specific manifest, but aging covers
    if manifest_id == 'test-aging':
        odv = vc.get('ordering_dependency_version', {}).get('Person', {}).get('age')
        if odv != 3:
            errors.append(f"ordering_dependency_version.Person.age = {odv}, expected 3")

    # Test Vector 7 (under-declared) – not in golden file, but engine should reject IR.
    # We cannot check from golden file; validator assumes IR was accepted.

    # Test Vector 8 (overflow) – not present in normal golden files; separate test.

    # Additional generic checks
    if manifest_id == 'test-divzero':
        if 'error' not in engine_obj:
            errors.append("Missing error record for division by zero manifest")
        else:
            err = engine_obj['error']
            if err.get('type') != 'DivisionByZero':
                errors.append(f"error.type = {err.get('type')}, expected DivisionByZero")
            if err.get('tick_number') != 1:
                errors.append(f"error.tick_number = {err.get('tick_number')}, expected 1")
            state = err.get('state_at_error', {})
            if 'Counter:0' not in state.get('nodes', {}):
                errors.append("state_at_error missing Counter:0")
            # Check version counters in error state
            vc_err = state.get('version_counters', {})
            if vc_err.get('extent_version', {}).get('Counter') != 1:
                errors.append("extent_version.Counter in error state != 1")
    return errors

# Comparison function
# ----------------------------------------------------------------------
def compare_with_canonical(engine_parsed: Dict, canonical_str: str) -> Tuple[bool, Optional[str]]:
    engine_canon = canonicalize_json(strip_non_normative(engine_parsed))
    if engine_canon == canonical_str:
        return True, None
    return False, engine_canon

def run_single_manifest(manifest_id: str, engine_output_path: str) -> bool:
    print(f"Validating {manifest_id} against {engine_output_path}...")
    with open(engine_output_path, 'rb') as f:
        raw = f.read()
    try:
        engine_parsed = json.loads(raw.decode('utf-8'))
    except Exception as e:
        print(f"ERROR: failed to parse JSON: {e}")
        return False

    canonical_str = CANONICAL.get(manifest_id)
    if not canonical_str:
        print(f"ERROR: unknown manifest {manifest_id}")
        return False

    ok, engine_canon = compare_with_canonical(engine_parsed, canonical_str)
    if not ok:
        print(f"FAIL: {manifest_id} – output does not match canonical golden file")
        print("Expected SHA256:", hashlib.sha256(canonical_str.encode('utf-8')).hexdigest())
        print("Engine   SHA256:", hashlib.sha256(engine_canon.encode('utf-8')).hexdigest())
        # Show diff? Could print first 200 chars
        return False

    # Additional counter invariants (Appendix C)
    errors = check_counter_invariants(engine_parsed, manifest_id)
    if errors:
        print(f"FAIL: {manifest_id} – counter invariant errors:")
        for e in errors:
            print(f"  - {e}")
        return False

    print(f"PASS: {manifest_id}")
    return True

def run_all_manifests(engine_output_dir: str):
    mapping = {
        "test-basic": "basic_output.json",
        "test-edge": "edge_output.json",
        "test-aging": "aging_output.json",
        "test-delete": "delete_output.json",
        "test-divzero": "divzero_output.json",
    }
    all_passed = True
    for manifest_id, default_filename in mapping.items():
        path = os.path.join(engine_output_dir, default_filename)
        if not os.path.exists(path):
            print(f"SKIP: {manifest_id} – file not found: {path}")
            continue
        if not run_single_manifest(manifest_id, path):
            all_passed = False
    return all_passed

# ----------------------------------------------------------------------
# Main
# ----------------------------------------------------------------------
def main():
    parser = argparse.ArgumentParser(description="GRISP Canonical IR Conformance Validator")
    parser.add_argument("--manifest", choices=list(CANONICAL.keys()), help="Manifest id")
    parser.add_argument("--engine-output", help="Path to engine golden file JSON")
    parser.add_argument("--all", action="store_true", help="Run all manifests (requires --engine-output-dir)")
    parser.add_argument("--engine-output-dir", help="Directory containing engine outputs for all manifests")
    args = parser.parse_args()

    if args.all:
        if not args.engine_output_dir:
            print("ERROR: --all requires --engine-output-dir")
            sys.exit(2)
        success = run_all_manifests(args.engine_output_dir)
        sys.exit(0 if success else 1)
    elif args.manifest and args.engine_output:
        success = run_single_manifest(args.manifest, args.engine_output)
        sys.exit(0 if success else 1)
    else:
        parser.print_help()
        sys.exit(2)

if __name__ == "__main__":
    main()

# GRISP Canonical IR Conformance Pack v0.29v7

This package contains the normative conformance test suite for the GRISP Canonical IR Specification v0.29v7.

## Files

- `canonical-ir-schema.json` – JSON Schema for the Canonical IR format.
- `grisp_validator.py` – Python validator script (requires Python 3.6+).
- `test-basic.golden.json` – Expected golden output for Manifest 1.
- `test-edge.golden.json` – Expected golden output for Manifest 2.
- `test-aging.golden.json` – Expected golden output for Manifest 3.
- `test-delete.golden.json` – Expected golden output for Manifest 4.
- `test-divzero.golden.json` – Expected golden output for Manifest 5.
- `README.md` – This file.

## Requirements

- Python 3.6 or later (no external libraries needed).
- An engine that produces golden files in GRISP Canonical JSON format (IR-018).

## Usage

### Validate a Single Manifest

```bash
python grisp_validator.py --manifest test-basic --engine-output engine_output.json
```

The validator will compare the engine output against the embedded canonical golden file and also run Appendix C counter invariant checks.

### Validate All Manifests

Place your engine outputs in a directory with the default names:

- `basic_output.json`
- `edge_output.json`
- `aging_output.json`
- `delete_output.json`
- `divzero_output.json`

Then run:

```bash
python grisp_validator.py --all --engine-output-dir /path/to/outputs
```

### Schema Validation

You can validate a Canonical IR document against the schema using any JSON Schema validator (e.g., `jsonschema` Python library, `ajv` CLI, or online tools). Example with Python:

```bash
pip install jsonschema
python -c "import json, jsonschema; jsonschema.validate(json.load(open('manifest.json')), json.load(open('canonical-ir-schema.json')))"
```

## Interpreting Results

- `PASS: <manifest>` – engine output matches canonical golden file and passes counter invariants.
- `FAIL:` – either byte mismatch or counter invariant violation. The validator prints expected and actual SHA256 hashes and lists any counter errors.

## Customizing File Names

You can change the default file names in `grisp_validator.py` inside `run_all_manifests()` if needed.

## Normative References

- GRISP Canonical IR Specification v0.29v7 (Freeze Edition)
- GRISP Deterministic Kernel Specification v0.29
- IR-018 (Canonical Serialization) – defines GRISP Canonical JSON
- Appendix C (Counter Optimization Guidance) – defines additional invariants

## Support

For questions, refer to the specification or open an issue on the GRISP repository.
